cmake_minimum_required(VERSION 3.8)
project(dori_bt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(dori_msgs REQUIRED)
add_executable(dori_tree_executor src/dori_bt_main.cpp)
target_include_directories(dori_tree_executor PRIVATE include)
ament_target_dependencies(dori_tree_executor rclcpp rclcpp_action ament_index_cpp
  std_msgs std_srvs nav2_msgs behaviortree_cpp_v3 dori_msgs)
target_link_libraries(dori_tree_executor nlohmann_json::nlohmann_json)
install(TARGETS dori_tree_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
ament_package()
